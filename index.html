🏨 Connect Your Database

One-time setup. Run the SQL below in Supabase, then enter your credentials.

  1. Go to supabase.com → sign up free → create a new project
  2. Open SQL Editor → paste and run the SQL block below
  3. Go to Project Settings → API → copy your Project URL and anon key
  4. Go to Authentication → Users → invite your staff members
CREATE TABLE IF NOT EXISTS rooms ( id INTEGER PRIMARY KEY, name TEXT, category TEXT, type TEXT, price INTEGER, emoji TEXT, max_guests INTEGER, status TEXT DEFAULT 'available', guest TEXT, checkin DATE, checkout DATE, source TEXT ); CREATE TABLE IF NOT EXISTS bookings ( id BIGSERIAL PRIMARY KEY, room_id INTEGER, room_name TEXT, guest TEXT, source TEXT, checkin DATE, checkout DATE, nights INTEGER, total_amount INTEGER, status TEXT DEFAULT 'confirmed', notes TEXT, staff_email TEXT, created_at TIMESTAMPTZ DEFAULT NOW(), checked_out_at TIMESTAMPTZ ); CREATE TABLE IF NOT EXISTS activity_log ( id BIGSERIAL PRIMARY KEY, action TEXT, room_id INTEGER, room_name TEXT, guest TEXT, source TEXT, date TEXT, checkout TEXT, staff_email TEXT, amount INTEGER, created_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE rooms ENABLE ROW LEVEL SECURITY; ALTER TABLE bookings ENABLE ROW LEVEL SECURITY; ALTER TABLE activity_log ENABLE ROW LEVEL SECURITY; CREATE POLICY "staff" ON rooms FOR ALL TO authenticated USING (true) WITH CHECK (true); CREATE POLICY "staff" ON bookings FOR ALL TO authenticated USING (true) WITH CHECK (true); CREATE POLICY "staff" ON activity_log FOR ALL TO authenticated USING (true) WITH CHECK (true);
